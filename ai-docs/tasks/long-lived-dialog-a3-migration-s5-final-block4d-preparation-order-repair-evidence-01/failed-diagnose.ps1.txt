$navigation=@($baseline.navigation|ForEach-Object {$now=Get-S54Pin $_.path 8388608;Assert-S54PinEqual $_ $now;$now})

for($n=0;$n -lt 7;$n++){
 $before=$preimages.images[$n];$null=Read-S54Pin $before.saved 134217728

 $stagePath=Join-Path $public ('image-preimages/staged-{0:d2}.json' -f $n)

 $null=Read-S54Pin $stage.destination 134217728
 $sourceNow=Get-S54Pin $stage.source.path 134217728
 $stages+=@{record=(Get-S54Pin $stagePath 65536);original=$before.original;saved=$before.saved;staged=$stage.destination;compiler_source_recorded=$stage.source;compiler_source_now=$sourceNow;compiler_source_still_matches=($sourceNow.bytes -eq $stage.source.bytes -and $sourceNow.sha256 -ceq $stage.source.sha256)}
}

 foreach($artifact in $capture.stable_artifacts){
  $normal=($name -ceq 'normal.json')
  $unit=@($capture.units|Where-Object {$_.package -ceq $artifact.package -and $_.target -ceq $artifact.target -and (@($_.kind)-join ',') -ceq $artifact.kind -and $_.profile.test -ne $normal -and $_.class -ceq 'runtime-target' -and $_.executable})

  $now=Get-S54Pin $unit[0].executable 134217728
  $captures+=@{capture=$name;package=$artifact.package;target=$artifact.target;kind=$artifact.kind;profile_test=$unit[0].profile.test;staged=$artifact;compiler_executable_now=$now;actual_compiler_bytes_match=($now.bytes -eq $artifact.bytes -and $now.sha256 -ceq $artifact.sha256)}
 }
}
