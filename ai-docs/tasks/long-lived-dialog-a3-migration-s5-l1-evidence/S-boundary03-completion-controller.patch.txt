+fn preserved_public_binding(base: &Path, evidence: &Path) -> Result<Vec<Value>> {
+    let observation: Value = serde_json::from_slice(&std::fs::read(
+        evidence.join("F-exhausted-boundary-03-observation.json"),
+    )?)?;
+    let root = base.canonicalize()?;
+    let public_root = evidence.canonicalize()?;
+    let mut bindings = Vec::new();
+    for entry in observation["exports"]
+        .as_array()
+        .context("case03 public exports")?
+    {
+        let source = workspace()
+            .join(entry["source"]["path"].as_str().context("source path")?)
+            .canonicalize()?;
+        let export = workspace()
+            .join(entry["export"]["path"].as_str().context("export path")?)
+            .canonicalize()?;
+        ensure!(
+            source.starts_with(&root) && export.starts_with(&public_root),
+            "bound public roots"
+        );
+        let bytes = std::fs::read(&source)?;
+        ensure!(
+            bytes == std::fs::read(&export)?,
+            "case03 public bytes changed"
+        );
+        bindings.push(json!({"source":source,"export":export,"bytes":bytes.len(),
+            "blake3":blake3::hash(&bytes).to_hex().to_string(),
+            "approved_export_sha256":entry["export"]["sha256"]}));
+    }
+    ensure!(
+        bindings.len() == 22,
+        "exact completed case03 public export set"
+    );
+    Ok(bindings)
